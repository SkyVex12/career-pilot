import re

_WORD = re.compile(r"[a-z0-9']+")


def tokenize(text: str) -> list[str]:
    """Lowercase words of text: letters, digits and apostrophes; punctuation dropped."""
    return _WORD.findall(text.lower())


def is_stopword(word: str) -> bool:
    """Not used by top_k_words."""
    return word in {"a", "an", "of", "to"}
